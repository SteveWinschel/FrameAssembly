// IPs can be defined like/as variables
LET example_client = 10.0.0.1 
LET google_dns = 8.8.8.8 

LET switch_agent = 10.0.10.5
LET example_nms = 10.0.10.100

// Templates are like functions, they can be called with arguments and are/should be reusable.
LET tcp_handshake(client, server) { 
    // The frame definitions are meant to look like they do in Wireshark
    client -> server TCP SYN SEQ=1 WIN=100 PAYLOAD="hello" WAIT=10ms 
    client <- server TCP ACK SEQ=2 WIN=200 PAYLOAD="world" WAIT=3m 
    client -> server TCP SYN ACK SEQ=3 WAIT=100s 
}

// Simple UDP template for sending simulated SNMP Traps
LET snmp_trap(agent, nms) {
    agent -> nms UDP PAYLOAD="RAW_SNMP_PAYLOAD_STRING" WAIT=10ms
}

// Alternatively you could define a "compile" block to generate a PCAP but you couldn't/shouldn't use an infinite loop for obvious reasons
COMPILE { 
    snmp_trap(switch_agent:161, example_nms:162)
    tcp_handshake(example_client:1234, google_dns:53) 
}
